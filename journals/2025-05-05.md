- #### Use "swayimg"
    - `wl-paste --type image/png | swayimg -` # From clipboard
        - `swayimg "exec://curl -sL https://example.com/sample.png"` # From a URL to an image
        - `curl -sL https://example.com/sample.png | swayimg -` # Method two
- ***Notes***
    - `vim ~/.config/swayimg/init.lua` # Configure as needed
      ```
      -- Copy image to clipboard
      swayimg.viewer.on_key("c", function()
        local file = os.tmpname() .. ".png"
        -- Export the currently displayed frame as PNG.
        swayimg.viewer.export(file)
        -- Copy the PNG to the Wayland clipboard.
        os.execute("wl-copy --type image/png < " .. string.format("%q", file))
        -- Remove the temporary file.
        os.remove(file)
      end)

      -- move image across the window
      swayimg.viewer.on_key("h", function()
        local pos = swayimg.viewer.get_position()
        swayimg.viewer.set_abs_position(pos.x + 10, pos.y)
      end)

      swayimg.viewer.on_key("j", function()
        local pos = swayimg.viewer.get_position()
        swayimg.viewer.set_abs_position(pos.x, pos.y - 10)
      end)

      swayimg.viewer.on_key("k", function()
        local pos = swayimg.viewer.get_position()
        swayimg.viewer.set_abs_position(pos.x, pos.y + 10)
      end)

      swayimg.viewer.on_key("l", function()
        local pos = swayimg.viewer.get_position()
        swayimg.viewer.set_abs_position(pos.x - 10, pos.y)
      end)

      -- exit from application
      swayimg.viewer.on_key("q", function()
        swayimg.exit()
      end)

      -- Hide text overlay by default on startup
      swayimg.text.visible = false
      ```
    - `f` # full screen
    - `t` # Text information
    - `-` # Zoom -10 (Opposite: `=`)
