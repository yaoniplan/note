- #### Use "niri"
    - Install dependencies
        - `doas yay --sync niri`
        - `doas pacman --sync alacritty fuzzel waybar swaybg swaylock otf-font-awesome`
    - Quit the current WM or DE, then select Niri Session (Instead of current using WM or DE)
- ***Notes***
    - Because Niri is more efficient than Hyprland.
        - Hyprland will cause the computer fan to spin wildly
    - `Super-v` # Toggle between floating and tiling
        - `Super-Shift-h/j/k/l` # Move window (Prevent floating windows from obscuring the content of tiled windows)
    - `Super-Shift-p` # Power off the monitor (sleep display)
    - `niri msg outputs` # View monitor name (e.g. LVDS-1) and resolution (e.g. 1366x768)
    - `niri --version` # Print version
    - Use scratchpad
        - `nix profile install github.com:argosnothing/niri-scratchpad-rs` # Install dependencies
        - `vim ~/.config/niri/config.kdl`
          ```
          // Set matched window to floating
          window-rule {
              match app-id "^chrome-.*-Default$"
              match app-id "^note$"
              open-floating true
          }
          // Spawn if not running
          Mod+Space { spawn "niri-scratchpad" "target" "--spawn" "alacritty --class note -e vim -c VimwikiMakeDiaryNote" "appid" "note"; }
          // Define a static workspace named `stash`
          workspace "stash" { }
          ```
        - https://github.com/argosnothing/niri-scratchpad-rs
    - `Alt-Tab` # Switch windows
        - `niri msg -j windows` # Get a list of open windows
        - `niri msg action focus-window --id 12` # Focus a window by id
        - `niri msg action focus-window-previous` # Focus previous window
    - `Super-o` # Toggle the overview (Or move your cursor to the top-left corner of the screen)
    - `Super-2` # Go to workspace 2 (In the overview mode)
    - `Super-c` # Center the column window
    - `Alt-Fn-PrtScn` # Capture the active window
    - `Super-Ctrl-h` # Move column window left
        - `Super-Ctrl-2` # Move column window to workspace 2
    - `Super-q` # Quit focused window
    - `Super-t` # Open terminal
    - `Super-f` # Maximize column (Toggle)
        - `Super-Shift-f` # Fullscreen window (Or `Super-m`)
    - `Super-[` # Move window left (Opposite: `Super-]`)
    - `Super-Shift-e` # Exit Niri (To test whether the hidden feature works at startup)
    - `Super-r` # Toggle between preset column widths (e.g. one third, one half and two thirds, opposite: `Super-Shift-r`)
        - `Super-Shift-=` # Increase column height by 10% (Opposite: `Super-Shift--`)
    - `vim ~/.config/niri/config.kdl` # Configure as needed
      ```
      // Make mpv config profile (pause_on_focus_lost_video) effective when switching Niri workspace
      debug {
          deactivate-unfocused-windows
      }
      // Enable a focus-ring (And disable border) separately for a specific application
      window-rule {
          match app-id="^dropdown$"

          border { off }

          focus-ring {
              on
              width 3
              active-color "#7fc8ff"
              inactive-color "#505050"
          }
      }
      layout {
          // If the workspace has only one window, center it.
          always-center-single-column
      }
      // Enable blur
      window-rule {
          exclude app-id="^mpv$"
          opacity 0.8
          background-effect {
              xray true
              blur true
          }
      }
      // Input device configuration.
      input {
          // Make the mouse warp to the center of newly focused windows (Even if the mouse is outside the window)
          warp-mouse-to-focus mode="center-xy-always"
      }
      // Animation settings.
      animations {
          // Turn off all animations to make it faster
          off
      }
      // Set corner radius of the highlight in the recent windows switcher
      recent-windows {
          highlight {
              corner-radius 14
          }
      }
      // Open floating windows with 1/2 of the screen width by default
      window-rule {
          match app-id=r#"^chrome-127\.0\.0\.1__-Default$"#
          //default-column-width { fixed 677; }
          //default-window-height { fixed 748; }
          default-column-width { proportion 0.5; }
          default-window-height { proportion 0.98; }
      }
      // Open tiling windows with 1/3 of the screen width by default
      window-rule {
          match app-id="^kitty$"
          default-column-width { proportion 0.33333; }
      }
      // Prevent the focus ring from being rendered as a solid background rectangle (Make swayimg work properly)
      window-rule {
          draw-border-with-background false
      }
      // Hide the important hotkeys pop-up at the start
      hotkey-overlay {
          skip-at-startup
      }
      // Uncomment to enable rounded corners
      window-rule {
          geometry-corner-radius 12
          clip-to-geometry true
          // The exclude directive is particularly suitable for special cases.
          exclude app-id="^dropdown$"
      }
      // Uncomment to disable title bar
      prefer-no-csd
      // Run at boot time
      spawn-at-startup "~/.local/bin/runAtBootTime.sh"
      binds {
          // Enable fake fullscreen
          Mod+Ctrl+Shift+F { toggle-windowed-fullscreen; }
          // Vim-like navigation (Try commenting or modifying the shortcut keys if there is a conflict)
          Mod+J { focus-window-or-workspace-down; }
          Mod+K { focus-window-or-workspace-up; }
          Mod+Shift+H { move-column-left; }
          Mod+Shift+J { move-window-down-or-to-workspace-down; }
          Mod+Shift+K { move-window-up-or-to-workspace-up; }
          Mod+Shift+L { move-column-right; }
          // Switch previous window
          Mod+Tab { focus-window-previous; }
          // Use parameters
          Mod+T { spawn "alacritty" "-e" "tmux"; }
          // Change launcher
          //Mod+D { spawn "sh" "-c" "tofi-run | sh"; }
          Mod+D { spawn-sh "tofi-run | sh"; }
      }

      // Include other configuration files
      // Since it's executed sequentially, remember to put it at the bottom.
      include "./colors.kdl"
      ```
