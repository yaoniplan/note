        - `--config ~/.local/src/ani/streamlink.conf` # Load custom config file
    - `--hls-start-offset 2:00` # Skip the first 2 minutes (Recommended, because it loads faster)
        - `--player-passthrough hls` # Hand the raw HLS URL directly to mpv (instead of piping data)
        - `--player-args '--start 00:02:00'` # Combine this option to achive the function of skipping the opening theme (But it will be slower)
    - `--stream-segment-threads 3` # Use 3 threads to download segments in parallel
    - `--player-no-close` # Prevent Streamlink from terminating the player when the stream ends
    - `--player-args '--speed=2'` # Pass arguments to the player
        - `--player-args "--include=\"$HOME/.local/src/ani/mpv.conf\" --force-media-title=\"$vod_name / $vod_title\""`
          ```
          # Wrap all mpv args in a double quoted string to solve the problem
          ```
