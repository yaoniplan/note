- #### Use "python"
    - `:.2f` # Format a number as a floating-point with 2 decimal places
        - `:` # The start of the format specifier
        - `.2` # Set the precision to 2 digits after the decimal point
        - `f` # Format as a float (fixed-point)
    - `print(real_url, end='')` # Output without a trailing newline
        - `, end=''` # Set the ending string to an empty string (As an input parameter for the next operation)
    - `url = url.replace("http://", "https://", 1)` # Replace http:// with https://
        - `1` # The first one
    - `subprocess.Popen(["openurl", f"vlc://{url}"])` # Open the subprocess to execute this command
        - `Popen()` # Process open
        - `[]` # Define a Python list
        - `f` # Formatted (Insert the value of a variable into the string)
        - `{}` # Evaluate the expression inside, convert it to string, and inject it into the spot
        - `([ ... ], start_new_session=True)` # Prevent command from exiting after the script has exited
