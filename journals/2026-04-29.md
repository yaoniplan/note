- #### Use "matugen"
    - `matugen image /tmp/wallhaven-4g8o8q.jpg --source-color-index 0` # Generate a color scheme (Place it after the image switching command)
    - `vim ~/.config/matugen/config.toml` # Configure as needed
      ```
      [config]

      [config.wallpaper]
      # Set wallpaper automatically while generating a color scheme
      set = true
      command = "awww img --transition-type any {{ image }}"

      [templates.niri]
      input_path = '~/.config/matugen/templates/niri-colors.kdl'
      output_path = '~/.config/niri/colors.kdl'
      post_hook = 'niri msg action load-config-file'
      
      [templates.mako]
      input_path = '~/.config/matugen/templates/mako'
      output_path = '~/.config/mako/mako-colors'
      post_hook = 'makoctl reload'
      ```
    - Niri
        - `vim ~/.config/matugen/templates/niri-colors.kdl`
          ```
          layout {
              background-color "transparent"
          
              focus-ring {
                  active-color   "{{colors.primary.default.hex}}"
                  inactive-color "{{colors.outline.default.hex}}"
                  urgent-color   "{{colors.error.default.hex}}"
              }
          
              border {
                  active-color   "{{colors.primary.default.hex}}"
                  inactive-color "{{colors.outline.default.hex}}"
                  urgent-color   "{{colors.error.default.hex}}"
              }
          
              shadow {
                  color "{{colors.shadow.default.hex}}70"
              }
          
              tab-indicator {
                  active-color   "{{colors.primary.default.hex}}"
                  inactive-color "{{colors.outline.default.hex}}"
                  urgent-color   "{{colors.error.default.hex}}"
              }
          
              insert-hint {
                  color "{{colors.primary.default.hex}}80"
              }
          }
          
          // Set colors for application with focus ring
          window-rule {
              focus-ring {
                  active-color "{{colors.primary.default.hex}}"
                  inactive-color "{{colors.outline.default.hex}}"
                  urgent-color "{{colors.error.default.hex}}"
              }
          }
          ```
        - `vim ~/.config/niri/config.kdl`
          ```
          layout {
              focus-ring {
                  off
              }
              background-color "transparent"
              border {
                  width 3
              }
              shadow {}
          }
          
          // Enable a focus-ring (And disable border) separately for a specific application
          window-rule {
              match app-id="^dropdown$"
              
              border { off }
              
              focus-ring {
                  on
                  width 3
              }
          }

          include "./colors.kdl"
          ```
    - Mako
        - `vim ~/.config/matugen/templates/mako`
          ```
          # Material You colors
          
          background-color={{colors.on_primary.default.hex}}
          text-color={{colors.tertiary.default.hex}}
          border-color={{colors.tertiary_container.default.hex}}
          # Per-mode settings
          
          [mode=do-not-disturb]
          invisible=1
          on-notify=none
          
          [mode=silent]
          on-notify=none
          
          [urgency=high]
          border-color={{colors.error_container.default.hex}}
          ```
        - `vim ~/.config/mako/config`
          ```
          ...
          # Add this line to the last line,
          # and there cannot be any syntax like`[xxx=xxx]` before it,
          # otherwise it will cause an error
          # `Unable to parse configuration file (fr.emersion.Mako.InvalidConfig)`
          include=~/.config/mako/mako-colors
          ```
