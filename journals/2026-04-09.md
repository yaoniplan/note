- #### Use "podman"
    - `vim Dockerfile` # Define the container image
      ```
      FROM python:3.12-alpine
      
      RUN apk add --no-cache git ffmpeg
      
      RUN git clone https://github.com/4ft35t/yt-dlp.git && \
          cd yt-dlp && \
          git checkout fix-ximalaya-album
      ```
    - `vim Makefile` # Build and run the container automatically
      ```
      IMAGE=yt-dlp-test
      PROXY=http://host.docker.internal:7890

      build:
          podman build --tag $(IMAGE_NAME) .

      shell:
          podman run --rm --interactive --tty \
            --volume "$(PWD):/workspace" \
            --wordir /yt-dlp \
            --env http_proxy=$(PROXY) \
            --env https_proxy=$(PROXY) \
            $(IMAGE) \
            sh

      clean:
          podman rmi $(IMAGE) || true
      ```
        - `--volume` # Mount a volume into the container (Similar to a shared directory)
        - `--workdir` # Set the default working directory inside the container
        - `--env` # Set environment variables in container
        - `make build` # Build the image (Contenient to reuse image)
        - `make shell` # Run that command into the shell environment
        - `./yt-dlp.sh "https://example.com/sample.mp3"` # Download to `/yt-dlp`
        - `cp sample.mp3 /workspace` # Copy to `/workspace` (Shared directory)
        - Use `Tab` (Instead of spaces) to solve the problem
          ```
          Makefile:17: *** missing separator.  Stop.
          ```
- ***Notes***
    - `podman images` # List images (`--all`)
    - `podman start qinglong` # Start the container
    - `podman stop qinglong || podman kill qinglong` # Stop (Or kill) the running container
    - `podman run -d --name qinglong -p 5700:5700 -v ~/.config/qinglong:/ql/data docker.io/whyour/qinglong:debian` # Run a container in the background
    - `podman exec qinglong task Cp0204_ChinaTelecomMonitor/telecom_monitor.py now` # Execute a command inside a running container
        - `env --unset=http_proxy --unset=https_proxy` # Remove variable from the environment (Place the command after container name)
    - Enable proxy (For containers)
      ```
      -e http_proxy=http://host.docker.internal:7890
      -e https_proxy=http://host.docker.internal:7890
      ```
    - `doas pacman --sync --refresh podman` # Install dependencies
    - Advantages: rootless, daemonless
    - Because I need to test some commands, but I don't want to pollute the host machine (environment isolation).

      git clone https://github.com/4ft35t/yt-dlp.git && cd yt-dlp && git checkout fix-ximalaya-album
      ./yt-dlp.sh --simulate https://www.ximalaya.com/album/4555085
