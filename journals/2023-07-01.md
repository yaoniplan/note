- #### Use "nmcli"
    - `nmcli device wifi connect yourSSID password yourPassword` # Connect a wifi
        - `nmcli device wifi list` # Get the SSID
    - `nmcli device wifi hotspot ssid yaoniplan password yaoniplan` # Create and start a hotspot
        - If your computer has a wireless network card (Wi-Fi adapter)
        - `nmcli connection up Hotspot` # Connect manually (After rebooting)
        - `nmcli connection show Hotspot` # Check the value of `connection.autoconnect`
        - `nmcli connection modify Hotspot autoconnect yes` # Enable autoconnect
        - `nmcli connection delete bfc2c6fa-eb8c-472c-b278-5953ec224e33` # Delte the hotspot
    - `nmcli connection down yourUUID` # Disconnect
        - `nmcli connection show` # Get the yourUUID
    - `nmcli connection up eno1` # Connect to a wired network
        - `nmcli device status` # Get the DEVICE
